cmake_minimum_required(VERSION 3.16)
project(rokae_fixed_impedance LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROKAE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" SDK_ARCH)
if(SDK_ARCH STREQUAL "amd64")
  set(SDK_ARCH x86_64)
elseif(SDK_ARCH STREQUAL "arm64")
  set(SDK_ARCH aarch64)
endif()
if(NOT SDK_ARCH MATCHES "^(x86_64|aarch64)$")
  message(FATAL_ERROR "Unsupported SDK architecture: ${SDK_ARCH}")
endif()
set(SDK_LIBRARY "${ROKAE_ROOT}/sdk/lib/${SDK_ARCH}/libxCoreSDK.so.0.3.4")
if(NOT EXISTS "${SDK_LIBRARY}")
  message(FATAL_ERROR "Missing SDK 0.3.4 runtime: ${SDK_LIBRARY}")
endif()
find_package(Threads REQUIRED)
add_executable(fixed_impedance main.cpp)
target_include_directories(fixed_impedance PRIVATE "${ROKAE_ROOT}/sdk/include")
target_compile_options(fixed_impedance PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(fixed_impedance PRIVATE "${SDK_LIBRARY}" Threads::Threads)
set_target_properties(fixed_impedance PROPERTIES BUILD_RPATH "${ROKAE_ROOT}/sdk/lib/${SDK_ARCH}")
add_executable(guard_test guard_test.cpp)
target_include_directories(guard_test PRIVATE "${ROKAE_ROOT}/sdk/include")
enable_testing()
add_test(NAME fixed_impedance_guard COMMAND guard_test)
add_executable(sdk_coordinate_test sdk_coordinate_test.cpp)
target_include_directories(sdk_coordinate_test PRIVATE "${ROKAE_ROOT}/sdk/include")
target_link_libraries(sdk_coordinate_test PRIVATE "${SDK_LIBRARY}")
set_target_properties(sdk_coordinate_test PROPERTIES BUILD_RPATH "${ROKAE_ROOT}/sdk/lib/${SDK_ARCH}")
add_test(NAME fixed_impedance_sdk_coordinates COMMAND sdk_coordinate_test)
add_executable(read_tools read_tools.cpp)
target_include_directories(read_tools PRIVATE "${ROKAE_ROOT}/sdk/include")
target_link_libraries(read_tools PRIVATE "${SDK_LIBRARY}")
set_target_properties(read_tools PROPERTIES BUILD_RPATH "${ROKAE_ROOT}/sdk/lib/${SDK_ARCH}")
add_executable(read_frames read_frames.cpp)
target_include_directories(read_frames PRIVATE "${ROKAE_ROOT}/sdk/include")
target_link_libraries(read_frames PRIVATE "${SDK_LIBRARY}")
set_target_properties(read_frames PROPERTIES BUILD_RPATH "${ROKAE_ROOT}/sdk/lib/${SDK_ARCH}")
